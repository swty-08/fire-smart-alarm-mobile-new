const Color _background = Color(0xFFFFF9F8);
const Color _cardColor = Color(0xFFFFF0EE);
const Color _statusCardColor = Color(0xFFFFDDE2);
const Color _accent = Color(0xFF9E5148);

class DashboardScreen extends StatefulWidget {
  const DashboardScreen({super.key});

  State<DashboardScreen> createState() => _DashboardScreenState();

class _DashboardScreenState extends State<DashboardScreen> {
  final ReadingRepository _repository = ReadingRepository();
  int _mobileTab = 0;
  int _desktopTab = 0;
  int _desktopHistoryCount = 20;
  DateTimeRange? _reportDateRange;
  String _reportStatus = 'Semua';
  String _selectedDetailDevice = 'Semua perangkat';
  String _selectedHistoryDevice = 'Semua perangkat';
  int _reportVisibleCount = 40;

  String? _lastNotifiedStatus;
  String? _lastNotifiedDevice;
  Future<void> _pickReportDateRange() async {
    final now = DateTime.now();
    final initialStart = _reportDateRange?.start ?? DateTime(now.year, now.month, now.day);
    DateTime visibleMonth = DateTime(initialStart.year, initialStart.month);
    DateTime? start = _reportDateRange?.start;
    DateTime? end = _reportDateRange?.end;

    final picked = await showDialog<DateTimeRange>(
      context: context,
      barrierDismissible: true,
      builder: (dialogContext) => StatefulBuilder(
        builder: (context, setDialogState) {
          final firstDay = DateTime(visibleMonth.year, visibleMonth.month, 1);
          final daysInMonth = DateUtils.getDaysInMonth(visibleMonth.year, visibleMonth.month);
          // Sunday-first calendar, matching the familiar desktop calendar layout.
          final leading = firstDay.weekday % 7;
          final cells = ((leading + daysInMonth + 6) ~/ 7) * 7;
          String format(DateTime d) => '${d.day}/${d.month}/${d.year}';
          bool sameDay(DateTime? a, DateTime b) => a != null && a.year == b.year && a.month == b.month && a.day == b.day;
          bool inRange(DateTime d) => start != null && end != null && !d.isBefore(DateUtils.dateOnly(start!)) && !d.isAfter(DateUtils.dateOnly(end!));

          return Dialog(
            insetPadding: const EdgeInsets.symmetric(horizontal: 20, vertical: 24),
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(18)),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 350),
              child: Padding(
                padding: const EdgeInsets.all(16),
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Row(children: [
                      const Expanded(child: Text('Pilih periode laporan', style: TextStyle(fontSize: 16, fontWeight: FontWeight.w700, color: Color(0xFF352725)))),
                      IconButton(tooltip: 'Tutup', onPressed: () => Navigator.pop(dialogContext), icon: const Icon(Icons.close_rounded, size: 20)),
                    ]),
                    Row(children: [
                      IconButton(onPressed: () => setDialogState(() => visibleMonth = DateTime(visibleMonth.year, visibleMonth.month - 1)), icon: const Icon(Icons.chevron_left_rounded)),
                      Expanded(child: Center(child: Text('${_monthName(visibleMonth.month)} ${visibleMonth.year}', style: const TextStyle(fontSize: 14, fontWeight: FontWeight.w600)))),
                      IconButton(onPressed: () => setDialogState(() => visibleMonth = DateTime(visibleMonth.year, visibleMonth.month + 1)), icon: const Icon(Icons.chevron_right_rounded)),
                    ]),
                    const SizedBox(height: 4),
                    Row(children: ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'].map((d) => Expanded(child: Center(child: Text(d, style: const TextStyle(fontSize: 10, color: Colors.black54, fontWeight: FontWeight.w600))))).toList()),
                    const SizedBox(height: 6),
                    GridView.builder(
                      shrinkWrap: true,
                      physics: const NeverScrollableScrollPhysics(),
                      itemCount: cells,
                      gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(crossAxisCount: 7, mainAxisSpacing: 3, crossAxisSpacing: 3),
                      itemBuilder: (context, index) {
                        final day = index - leading + 1;
                        if (day < 1 || day > daysInMonth) return const SizedBox.shrink();
                        final date = DateTime(visibleMonth.year, visibleMonth.month, day);
                        final selected = sameDay(start, date) || sameDay(end, date);
                        final range = inRange(date);
                        return InkWell(
                          borderRadius: BorderRadius.circular(20),
                          onTap: () => setDialogState(() {
                            if (start == null || end != null || date.isBefore(start!)) {
                              start = date;
                              end = null;
                            } else {
                              end = date;
                            }
                          }),
                          child: Container(
                            decoration: BoxDecoration(color: selected ? _accent : range ? _cardColor : Colors.transparent, shape: BoxShape.circle),
                            alignment: Alignment.center,
                            child: Text('$day', style: TextStyle(fontSize: 12, color: selected ? Colors.white : const Color(0xFF352725), fontWeight: selected ? FontWeight.w700 : FontWeight.normal)),
                          ),
                        );
                      },
                    ),
                    const SizedBox(height: 10),
                    Align(alignment: Alignment.centerLeft, child: Text(start == null ? 'Pilih tanggal mulai' : end == null ? 'Mulai: ${format(start!)} · Pilih tanggal akhir' : '${format(start!)} – ${format(end!)}', style: const TextStyle(fontSize: 11, color: Colors.black54))),
                    const SizedBox(height: 12),
                    Row(mainAxisAlignment: MainAxisAlignment.end, children: [
                      TextButton(onPressed: () => Navigator.pop(dialogContext), child: const Text('Batal', style: TextStyle(fontSize: 13))),
                      const SizedBox(width: 6),
                      FilledButton(onPressed: start != null && end != null ? () => Navigator.pop(dialogContext, DateTimeRange(start: DateUtils.dateOnly(start!), end: DateUtils.dateOnly(end!))) : null, style: FilledButton.styleFrom(backgroundColor: _accent, padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 10)), child: const Text('Terapkan', style: TextStyle(fontSize: 13))),
                    ]),
                  ],
                ),
              ),
            ),
          );
        },
      ),
    );
    if (picked != null && mounted) {
      setState(() {
        _reportDateRange = picked;
        _reportVisibleCount = 40;
      });
    }
  }

  String _monthName(int month) {
    const months = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
    return months[month - 1];
  }

  Widget _buildLatestReadingsSection(List<Reading> deviceReadings) {
    final deviceIds = deviceReadings
        .map((reading) => reading.deviceId.trim().isEmpty
            ? 'Perangkat tanpa ID'
            : reading.deviceId.trim())
        .toSet()
        .toList()
      ..sort();

    final selectedDevice = _selectedDetailDevice == 'Semua perangkat' ||
            deviceIds.contains(_selectedDetailDevice)
        ? _selectedDetailDevice
        : 'Semua perangkat';

    final selectedReadings = selectedDevice == 'Semua perangkat'
        ? deviceReadings
        : deviceReadings.where((reading) {
            final id = reading.deviceId.trim().isEmpty
                ? 'Perangkat tanpa ID'
                : reading.deviceId.trim();
            return id == selectedDevice;
          }).toList();

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          children: [
            const Expanded(
              child: Text(
                'Detail Pembacaan Terbaru',
                style: TextStyle(
                  fontSize: 16,
                  fontWeight: FontWeight.bold,
                  color: Colors.black87,
                ),
              ),
            ),
            const SizedBox(width: 10),
            SizedBox(
              width: MediaQuery.of(context).size.width < 600 ? 180 : 200,
              child: DropdownButtonFormField<String>(
                value: selectedDevice,
                isExpanded: true,
                style: const TextStyle(fontSize: 12, color: Colors.black87),
                selectedItemBuilder: (context) => [
                  'Semua perangkat',
                  ...deviceIds,
                ].map((id) => Align(
                      alignment: Alignment.centerLeft,
                      child: Text(
                        id,
                        maxLines: 1,
                        overflow: TextOverflow.ellipsis,
                        style: const TextStyle(fontSize: 12, color: Colors.black87),
                      ),
                    )).toList(),
                decoration: InputDecoration(
                  isDense: true,
                  filled: true,
                  fillColor: Colors.white,
                  contentPadding: const EdgeInsets.symmetric(
                    horizontal: 10,
                    vertical: 10,
                  ),
                  border: OutlineInputBorder(
                    borderRadius: BorderRadius.circular(10),
                    borderSide: const BorderSide(color: _cardColor),
                  ),
                  enabledBorder: OutlineInputBorder(
                    borderRadius: BorderRadius.circular(10),
                    borderSide: const BorderSide(color: _cardColor),
                  ),
                ),
                items: [
                  const DropdownMenuItem(
                    value: 'Semua perangkat',
                    child: const Text(
                      'Semua perangkat',
                      style: TextStyle(fontSize: 12),
                      overflow: TextOverflow.ellipsis,
                    ),
                  ),
                  ...deviceIds.map((id) => DropdownMenuItem(
                        value: id,
                        child: Text(
                          id,
                          style: const TextStyle(fontSize: 12),
                          overflow: TextOverflow.ellipsis,
                        ),
                      )),
                ],
                onChanged: (value) {
                  if (value != null) {
                    setState(() => _selectedDetailDevice = value);
                  }
                },
              ),
            ),
          ],
        ),
        const SizedBox(height: 12),
        if (selectedReadings.isEmpty)
          const Text('Belum ada pembacaan sensor.')
        else
          ...selectedReadings.map((reading) {
            final deviceId = reading.deviceId.trim().isEmpty
                ? 'Perangkat tanpa ID'
                : reading.deviceId.trim();
            return Container(
              width: double.infinity,
              margin: const EdgeInsets.only(bottom: 14),
              padding: const EdgeInsets.all(12),
              decoration: BoxDecoration(
                color: Colors.white,
                borderRadius: BorderRadius.circular(18),
                border: Border.all(color: _cardColor),
                boxShadow: [
                  BoxShadow(
                    color: Colors.black.withValues(alpha: 0.06),
                    blurRadius: 5,
                    offset: const Offset(0, 2),
                  ),
                ],
              ),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Row(
                    children: [
                      const Icon(Icons.memory, color: _accent, size: 20),
                      const SizedBox(width: 7),
                      Expanded(
                        child: Text(
                          deviceId,
                          style: const TextStyle(
                            fontSize: 14,
                            fontWeight: FontWeight.bold,
                            color: Colors.black87,
                          ),
                        ),
                      ),
                      Text(
                        _formatDateTime(reading.timestamp),
                        style: const TextStyle(fontSize: 10, color: Colors.black54),
                      ),
                    ],
                  ),
                  const SizedBox(height: 10),
                  _buildStatusCard(reading),
                  const SizedBox(height: 10),
                  LayoutBuilder(
                    builder: (context, constraints) {
                      final compact = constraints.maxWidth < 420;
                      if (compact) {
                        return Column(
                          children: [
                            Row(children: [
                              Expanded(child: _buildSensorCard(
                                icon: Icons.thermostat,
                                title: 'Temperature',
                                value: '${reading.temperature.toStringAsFixed(1)} °C',
                              )),
                              const SizedBox(width: 8),
                              Expanded(child: _buildSensorCard(
                                icon: Icons.water_drop,
                                title: 'Humidity',
                                value: '${reading.humidity.toStringAsFixed(1)} %',
                              )),
                            ]),
                            const SizedBox(height: 8),
                            _buildMq2Card(reading),
                          ],
                        );
                      }
                      return Row(children: [
                        Expanded(child: _buildSensorCard(
                          icon: Icons.thermostat,
                          title: 'Temperature',
                          value: '${reading.temperature.toStringAsFixed(1)} °C',
                        )),
                        const SizedBox(width: 8),
                        Expanded(child: _buildSensorCard(
                          icon: Icons.water_drop,
                          title: 'Humidity',
                          value: '${reading.humidity.toStringAsFixed(1)} %',
                        )),
                        const SizedBox(width: 8),
                        Expanded(child: _buildMq2Card(reading)),
                      ]);
                    },
                  ),
                ],
              ),
            );
          }),
      ],
    );
  }

  @override
  Widget build(BuildContext context) {
    final isWeb = MediaQuery.sizeOf(context).width >= 900;
    return Scaffold(
      backgroundColor: _background,
      appBar: isWeb
          ? null
          : AppBar(
              backgroundColor: _background,
              elevation: 0,
              centerTitle: true,
              title: Text(
                const ['Beranda', 'Riwayat', 'Laporan', 'Pengaturan'][_mobileTab],
                style: const TextStyle(fontSize: 20, fontWeight: FontWeight.bold, color: Colors.black87),
              ),
              actions: const [
                Padding(
                  padding: EdgeInsets.only(right: 16),
                  child: Icon(Icons.notifications, color: Color(0xFFB33A35)),
                ),
              ],
            ),

      bottomNavigationBar: isWeb
          ? null
          : BottomNavigationBar(
              currentIndex: _mobileTab,
              selectedItemColor: _accent,
              unselectedItemColor: Colors.black54,
              backgroundColor: Colors.white,
              type: BottomNavigationBarType.fixed,
              onTap: (i) => setState(() => _mobileTab = i),
              items: const [
                BottomNavigationBarItem(icon: Icon(Icons.home_rounded), label: 'Beranda'),
                BottomNavigationBarItem(icon: Icon(Icons.list_alt_rounded), label: 'Riwayat'),
                BottomNavigationBarItem(icon: Icon(Icons.assessment_rounded), label: 'Laporan'),
                BottomNavigationBarItem(icon: Icon(Icons.settings_rounded), label: 'Pengaturan'),
              ],
            ),
      body: StreamBuilder<List<Reading>>(
        stream: _repository.getReadings(),
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting &&
              !snapshot.hasData) {
            return const Center(
              child: CircularProgressIndicator(color: _accent),
            );
          }
          if (snapshot.hasError) {
            return _message(
              'Data tidak dapat dibaca',
              'Periksa koneksi internet dan pengaturan Firebase.\n${snapshot.error}',
            );
          }
          final readings = snapshot.data ?? [];
          if (readings.isEmpty) {
            if (!isWeb && _mobileTab == 2) {
              return SafeArea(child: SingleChildScrollView(
                padding: const EdgeInsets.fromLTRB(14, 12, 14, 24),
                child: _buildReportSection(readings),
              ));
            }
            if (!isWeb && _mobileTab == 3) {
              return const MobileSettingsScreen(embedded: true);
            }
            if (!isWeb && _mobileTab == 1) {
              return _message('Belum ada riwayat', 'Data sensor akan tampil setelah tersedia di Firebase.');
            }
            return _message(
              'Belum ada data sensor',
              'Data dari Firebase akan tampil di halaman ini.',
            );
          }
          final historyDeviceIds = readings
              .map((reading) => reading.deviceId.trim().isEmpty
                  ? 'Perangkat tanpa ID'
                  : reading.deviceId.trim())
              .toSet()
              .toList()
            ..sort();
          final selectedHistoryDevice =
              _selectedHistoryDevice == 'Semua perangkat' ||
                      historyDeviceIds.contains(_selectedHistoryDevice)
                  ? _selectedHistoryDevice
                  : 'Semua perangkat';
          final historyDeviceReadings = selectedHistoryDevice == 'Semua perangkat'
              ? readings
              : readings.where((reading) {
                  final id = reading.deviceId.trim().isEmpty
                      ? 'Perangkat tanpa ID'
                      : reading.deviceId.trim();
                  return id == selectedHistoryDevice;
                }).toList();

          // Ambil pembacaan terbaru untuk SETIAP ESP32 berdasarkan deviceId.
          // Stream Firestore tetap satu, tetapi dashboard tidak lagi hanya
          // menggunakan readings.first (yang sebelumnya mewakili satu device).
          final latestByDevice = <String, Reading>{};
          for (final reading in readings) {
            final id = reading.deviceId.trim().isEmpty
                ? 'Perangkat tanpa ID'
                : reading.deviceId.trim();
            final previous = latestByDevice[id];
            if (previous == null || reading.timestamp.isAfter(previous.timestamp)) {
              latestByDevice[id] = reading;
            }
          }
          final deviceReadings = latestByDevice.values.toList()
            ..sort((a, b) => b.timestamp.compareTo(a.timestamp));
          final latest = readings.first;
          final currentStatus = latest.status.toLowerCase();

          if (currentStatus == 'warning' || currentStatus == 'danger') {
            if (_lastNotifiedStatus != latest.status ||
                _lastNotifiedDevice != latest.deviceId) {
              _lastNotifiedStatus = latest.status;
              _lastNotifiedDevice = latest.deviceId;

              WidgetsBinding.instance.addPostFrameCallback((_) {
                if (!mounted) return;

                NotificationService.showStatusNotification(
                  status: latest.status,
                  temperature: latest.temperature,
                  smoke: latest.smoke,
                );
              });
            }
          } else {
            _lastNotifiedStatus = null;
            _lastNotifiedDevice = null;
          }
          final trend = readings.take(20).toList().reversed.toList();
          if (!isWeb) {
            if (_mobileTab == 1) {
              final history = historyDeviceReadings.take(_desktopHistoryCount).toList();
              return SafeArea(child: ListView(
                padding: const EdgeInsets.fromLTRB(16, 12, 16, 24),
                children: [
                  _sectionTitle('Pilih perangkat'),
                  const SizedBox(height: 8),
                  SizedBox(
                    width: 280,
                    child: DropdownButtonFormField<String>(
                      value: selectedHistoryDevice,
                      isExpanded: true,
                      style: const TextStyle(fontSize: 12, color: Color(0xFF352725)),
                      decoration: InputDecoration(
                        isDense: true,
                        contentPadding: const EdgeInsets.symmetric(horizontal: 12, vertical: 10),
                        filled: true,
                        fillColor: const Color(0xFFFFFBFA),
                        border: OutlineInputBorder(borderRadius: BorderRadius.circular(10), borderSide: const BorderSide(color: Color(0xFF9E5148))),
                        enabledBorder: OutlineInputBorder(borderRadius: BorderRadius.circular(10), borderSide: const BorderSide(color: Color(0xFF9E5148))),
                      ),
                      items: [
                        const DropdownMenuItem(value: 'Semua perangkat', child: Text('Semua perangkat', style: TextStyle(fontSize: 12))),
                        ...historyDeviceIds.map((id) => DropdownMenuItem(value: id, child: Text(id, style: const TextStyle(fontSize: 12), overflow: TextOverflow.ellipsis))),
                      ],
                      onChanged: (value) {
                        if (value != null) setState(() => _selectedHistoryDevice = value);
                      },
                    ),
                  ),
                  const SizedBox(height: 16),
                  _sectionTitle('Pilih jumlah data'),
                  const SizedBox(height: 10),
                  Row(children: [20, 50, 100].map((count) => Expanded(child: Padding(
                    padding: EdgeInsets.only(right: count == 100 ? 0 : 8),
                    child: ChoiceChip(
                      label: Center(child: Text('$count')),
                      selected: _desktopHistoryCount == count,
                      showCheckmark: false,
                      selectedColor: _accent,
                      backgroundColor: _cardColor,
                      labelStyle: TextStyle(color: _desktopHistoryCount == count ? Colors.white : Colors.black87, fontWeight: FontWeight.bold),
                      onSelected: (_) => setState(() => _desktopHistoryCount = count),
                    ),
                  ))).toList()),
                  const SizedBox(height: 14),
                  _buildChartCard(history.reversed.toList(), height: 190),
                  const SizedBox(height: 20),
                  _sectionTitle('Riwayat Sensor (${history.length} data)'),
                  const SizedBox(height: 10),
                  if (history.isEmpty) const Text('Belum ada riwayat sensor.')
                  else ...history.asMap().entries.map((e) => _buildHistoryCard(e.value, e.key + 1)),
                ],
              ));
            }
            if (_mobileTab == 2) {
              return SafeArea(child: SingleChildScrollView(
                padding: const EdgeInsets.fromLTRB(14, 12, 14, 24),
                child: _buildReportSection(readings),
              ));
            }
            if (_mobileTab == 3) return const MobileSettingsScreen(embedded: true);
            return SafeArea(
              child: Center(
                child: ConstrainedBox(
                  constraints: const BoxConstraints(maxWidth: 560),
                  child: ListView(
                    padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
                    children: [
                      _buildLatestReadingsSection(deviceReadings),
                      const SizedBox(height: 12),
                      const _AlarmControlCard(),
                      const SizedBox(height: 20),
                      _sectionTitle('Tren 20 data terakhir'),
                      const SizedBox(height: 10),
                      _buildChartCard(trend, height: 180),
                      const SizedBox(height: 20),
                      _sectionTitle('Riwayat Sensor Terbaru'),
                      const SizedBox(height: 10),
                      ...readings
                          .take(5)
                          .toList()
                          .asMap()
                          .entries
                          .map((e) => _buildHistoryCard(e.value, e.key + 1)),
                      OutlinedButton.icon(
                        onPressed: () => setState(() => _mobileTab = 1),
                        icon: const Icon(Icons.history),
                        label: const Text('Lihat Semua Riwayat'),
                      ),
                    ],
                  ),
                ),
              ),
            );
          }
          return Row(
            children: [
              Container(
                width: 205,
                color: const Color(0xFF9E342F),
                child: SafeArea(
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      const Padding(
                        padding: EdgeInsets.fromLTRB(20, 22, 12, 30),
                        child: Row(
                          children: [
                            Icon(
                              Icons.local_fire_department,
                              color: Colors.white,
                              size: 30,
                            ),
                            SizedBox(width: 8),
                            Expanded(
                              child: Text(
                                'Fire Smart Alarm',
                                style: TextStyle(
                                  color: Colors.white,
                                  fontWeight: FontWeight.bold,
                                  fontSize: 16,
                                ),
                              ),
                            ),
                          ],
                        ),
                      ),
                      _sideItem(
                        Icons.dashboard,
                        'Dashboard',
                        _desktopTab == 0,
                        () => setState(() => _desktopTab = 0),
                      ),
                      _sideItem(
                        Icons.list_alt,
                        'Riwayat Sensor',
                        _desktopTab == 1,
                        () => setState(() => _desktopTab = 1),
                      ),
                      _sideItem(
                        Icons.show_chart,
                        'Grafik',
                        _desktopTab == 2,
                        () => setState(() => _desktopTab = 2),
                      ),
                      _sideItem(
                        Icons.settings,
                        'Pengaturan',
                        _desktopTab == 3,
                        () => setState(() => _desktopTab = 3),
                      ),
                      _sideItem(
                        Icons.info,
                        'Tentang',
                        _desktopTab == 4,
                        () => setState(() => _desktopTab = 4),
                      ),

                      _sideItem(
                        Icons.assessment,
                        'Laporan',
                        _desktopTab == 5,
                        () => setState(() => _desktopTab = 5),
                      ),

                      const SizedBox(height: 12),

                      _sideItem(Icons.logout, 'Logout', false, () async {
                        final shouldLogout = await showDialog<bool>(
                          context: context,
                          builder: (dialogContext) => AlertDialog(
                            title: const Text('Logout'),
                            content: const Text(
                              'Apakah kamu yakin ingin keluar dari akun?',
                            ),
                            actions: [
                              TextButton(
                                onPressed: () =>
                                    Navigator.pop(dialogContext, false),
                                child: const Text('Batal'),
                              ),
                              FilledButton(
                                onPressed: () =>
                                    Navigator.pop(dialogContext, true),
                                child: const Text('Logout'),
                              ),
                            ],
                          ),
                        );

                        if (shouldLogout == true) {
                          await FirebaseAuth.instance.signOut();
                        }
                      }),
                    ],
                  ),
                ),
              ),
              Expanded(
                child: SafeArea(
                  child: ListView(
                    padding: const EdgeInsets.fromLTRB(22, 18, 22, 26),
                    children: [
                      if (_desktopTab != 5) ...[
                        Row(
                          children: [
                            Expanded(
                              child: Column(
                                crossAxisAlignment: CrossAxisAlignment.start,
                                children: [
                                  Text(
                                    _desktopTab == 1
                                        ? 'Riwayat Sensor'
                                        : _desktopTab == 2
                                        ? 'Grafik Sensor'
                                        : _desktopTab == 3
                                        ? 'Pengaturan Sistem'
                                        : _desktopTab == 4
                                        ? 'Tentang Fire Smart Alarm'
                                        : 'Dashboard Monitoring',
                                    style: const TextStyle(
                                      fontSize: 21,
                                      fontWeight: FontWeight.bold,
                                      color: Colors.black87,
                                    ),
                                  ),
                                  const SizedBox(height: 5),
                                  Text(
                                    _desktopTab == 1
                                        ? 'Lihat riwayat pembacaan sensor.'
                                        : _desktopTab == 2
                                        ? 'Pantau perubahan suhu, kelembapan, dan asap.'
                                        : _desktopTab == 3
                                        ? 'Kelola alarm dan lihat aturan kondisi sensor.'
                                        : 'Pantau kondisi laboratorium secara real-time',
                                    style: const TextStyle(
                                      color: Colors.black54,
                                    ),
                                  ),
                                ],
                              ),
                            ),
                            const Icon(
                              Icons.circle,
                              color: Colors.red,
                              size: 10,
                            ),
                            const SizedBox(width: 7),
                            const Text(

  Widget _buildReportSection(List<Reading> readings) {
    final filtered = readings.where((reading) {
      final statusMatches =
          _reportStatus == 'Semua' ||
          reading.status.toLowerCase() == _reportStatus.toLowerCase();
      final range = _reportDateRange;
      final dateMatches =
          range == null ||
          (!reading.timestamp.isBefore(
                DateTime(range.start.year, range.start.month, range.start.day),
              ) &&
              reading.timestamp.isBefore(
                DateTime(range.end.year, range.end.month, range.end.day + 1),
              ));
      return statusMatches && dateMatches;
    }).toList();

    final dangerCount = filtered
        .where((r) => r.status.toLowerCase() == 'danger')
        .length;
    final warningCount = filtered
        .where((r) => r.status.toLowerCase() == 'warning')
        .length;
    final normalCount = filtered
        .where((r) => r.status.toLowerCase() == 'normal')
        .length;
    final averageTemp = filtered.isEmpty
        ? 0.0
        : filtered.map((r) => r.temperature).reduce((a, b) => a + b) /
              filtered.length;
    final averageHumidity = filtered.isEmpty
        ? 0.0
        : filtered.map((r) => r.humidity).reduce((a, b) => a + b) /
              filtered.length;
    final averageSmoke = filtered.isEmpty
        ? 0.0
        : filtered.map((r) => r.smoke).reduce((a, b) => a + b) /
              filtered.length;
    final chartReadings = filtered.reversed.toList();
    // Batasi titik grafik agar CustomPainter tidak menggambar ribuan titik.
    const maxChartPoints = 180;
    final chartStep = chartReadings.length > maxChartPoints
        ? (chartReadings.length / maxChartPoints).ceil()
        : 1;
    final chartPoints = chartStep == 1
        ? chartReadings
        : [
            for (var i = 0; i < chartReadings.length; i += chartStep)
              chartReadings[i],
          ];
    final visibleMobileCount = _reportVisibleCount.clamp(0, filtered.length);
    final screenWidth = MediaQuery.sizeOf(context).width;
    final isMobile = screenWidth < 600;
    final isNarrow = screenWidth < 1100;

    return Center(
      child: ConstrainedBox(
        constraints: const BoxConstraints(maxWidth: 1400),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Row(
              children: [
                const Expanded(
                  child: Text(
                    'Laporan Monitoring Sensor',
                    style: TextStyle(
                      fontSize: 23,
                      fontWeight: FontWeight.bold,
                      color: Colors.black87,
                    ),
                  ),
                ),
              ],
            ),
            const SizedBox(height: 5),
            const Text(
              'Ringkasan dan riwayat pembacaan sensor berdasarkan periode dan status.',
              style: TextStyle(color: Colors.black54, fontSize: 13),
            ),
            const SizedBox(height: 16),
            Container(
              padding: const EdgeInsets.all(18),
              decoration: BoxDecoration(
                color: Colors.white,
                borderRadius: BorderRadius.circular(18),
                border: Border.all(color: const Color(0xFFF0E2DF)),
                boxShadow: const [
                  BoxShadow(color: Color(0x0D6D332D), blurRadius: 18, offset: Offset(0, 5)),
                ],
              ),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Row(children: [
                    Container(
                      padding: const EdgeInsets.all(9),
                      decoration: BoxDecoration(color: _cardColor, borderRadius: BorderRadius.circular(12)),
                      child: const Icon(Icons.tune_rounded, color: _accent, size: 20),
                    ),
                    const SizedBox(width: 10),
                    const Expanded(child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
                      Text('Filter laporan', style: TextStyle(fontSize: 14, fontWeight: FontWeight.w700, color: Color(0xFF352725))),
                      SizedBox(height: 2),
                      Text('Sesuaikan periode dan status sensor', style: TextStyle(fontSize: 11, color: Colors.black54)),
                    ])),
                    if (!isMobile)
                      FilledButton.icon(
                        onPressed: filtered.isEmpty ? null : () => _exportReportPdf(filtered),
                        icon: const Icon(Icons.picture_as_pdf_rounded, size: 18),
                        label: const Text('Ekspor PDF'),
                        style: FilledButton.styleFrom(backgroundColor: _accent, foregroundColor: Colors.white, padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 13), textStyle: const TextStyle(fontSize: 14), shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12))),
                      ),
                  ]),
                  const SizedBox(height: 16),
                  Wrap(
                    spacing: 12,
                    runSpacing: 10,
                    crossAxisAlignment: WrapCrossAlignment.center,
                    children: [
                  OutlinedButton.icon(
                    onPressed: _pickReportDateRange,
                    style: OutlinedButton.styleFrom(foregroundColor: _accent, side: const BorderSide(color: Color(0xFFE3C9C4)), padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 15), shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12))),
                    icon: const Icon(Icons.calendar_month_rounded, size: 19),
                    label: Text(
                      _reportDateRange == null
                          ? 'Semua tanggal'
                          : '${_dateOnly(_reportDateRange!.start)} – ${_dateOnly(_reportDateRange!.end)}',

                  ),
                  if (isMobile)
                    FilledButton.icon(

                return Wrap(

                  runSpacing: gap,

                      width: width,

                        '${filtered.length}',

                        _accent,
                      ),

                    SizedBox(
                      width: width,
                      child: _reportSummaryCard(
                        'Normal',
                        '$normalCount',
                        Icons.check_circle,
                        Colors.green,
                      ),

                      width: width,

            const SizedBox(height: 12),
            LayoutBuilder(
              builder: (context, constraints) {

                const gap = 12.0;
                final width =
                    (constraints.maxWidth - gap * (columns - 1)) / columns;
                return Wrap(
                  spacing: gap,

                  children: [

                      width: width,

                    SizedBox(
                      width: width,

            const Text(

            if (filtered.length < 2)

      ),
    );
    await Printing.layoutPdf(

  }

String _dateOnly(DateTime date) =>
    '${date.day.toString().padLeft(2, '0')}/${date.month.toString().padLeft(2, '0')}/${date.year}';

class _ReportLinePainter extends CustomPainter {
  _ReportLinePainter(this.values, this.lineColor);
  final List<double> values;
  final Color lineColor;

  @override
  void paint(Canvas canvas, Size size) {
    if (values.length < 2 || size.isEmpty) return;
    const pad = 10.0;
    final minValue = values.reduce((a, b) => a < b ? a : b);
    final maxValue = values.reduce((a, b) => a > b ? a : b);
    final range = (maxValue - minValue).abs() < 0.0001
        ? 1.0
        : maxValue - minValue;
    final chartHeight = size.height - pad * 2;
    final gridPaint = Paint()
      ..color = const Color(0xFFE8DEDC)
      ..strokeWidth = 1;
    for (var i = 0; i < 4; i++) {
      final y = pad + chartHeight * i / 3;
      canvas.drawLine(Offset(0, y), Offset(size.width, y), gridPaint);
    }
    final path = Path();
    for (var i = 0; i < values.length; i++) {
      final x = size.width * i / (values.length - 1);
      final y = pad + chartHeight * (1 - (values[i] - minValue) / range);
      if (i == 0) {
        path.moveTo(x, y);
      } else {
        path.lineTo(x, y);
      }
    }
    canvas.drawPath(
      path,
      Paint()
        ..color = lineColor
        ..strokeWidth = 2.2
        ..style = PaintingStyle.stroke
        ..strokeCap = StrokeCap.round
        ..strokeJoin = StrokeJoin.round,
    );
  }

  @override
  bool shouldRepaint(covariant _ReportLinePainter oldDelegate) =>
      oldDelegate.lineColor != lineColor ||
      oldDelegate.values.length != values.length ||
      oldDelegate.values.asMap().entries.any(
        (entry) => entry.value != values[entry.key],
      );
}

class _ThresholdEditDialog extends StatefulWidget {
  const _ThresholdEditDialog({
    required this.currentWarning,
    required this.currentDanger,
  });

  final int currentWarning;
  final int currentDanger;

  @override
  State<_ThresholdEditDialog> createState() =>
      _ThresholdEditDialogState();
}

class _ThresholdEditDialogState extends State<_ThresholdEditDialog> {
  final _formKey = GlobalKey<FormState>();

  late final TextEditingController _warningController;
  late final TextEditingController _dangerController;

  @override
  void initState() {
    super.initState();
    _warningController = TextEditingController(
      text: widget.currentWarning.toString(),
    );
    _dangerController = TextEditingController(
      text: widget.currentDanger.toString(),
    );
  }

  @override
  void dispose() {
    _warningController.dispose();
    _dangerController.dispose();
    super.dispose();
  }

  void _save() {
    if (!_formKey.currentState!.validate()) return;

    Navigator.of(context).pop<Map<String, int>>({
      'thresholdWarning': int.parse(_warningController.text),
      'thresholdDanger': int.parse(_dangerController.text),
    });
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      title: const Text('Ubah Threshold Sensor'),
      content: Form(
        key: _formKey,
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            TextFormField(
              controller: _warningController,
              keyboardType: TextInputType.number,
              decoration: const InputDecoration(
                labelText: 'Threshold Warning',
                suffixText: 'nilai MQ-2',
              ),
              validator: (value) {
                final parsed = int.tryParse(value ?? '');
                if (parsed == null || parsed < 1 || parsed > 4095) {
                  return 'Masukkan angka 1 sampai 4095';
                }
                return null;
              },
            ),
            const SizedBox(height: 12),
            TextFormField(
              controller: _dangerController,
              keyboardType: TextInputType.number,
              decoration: const InputDecoration(
                labelText: 'Threshold Danger',
                suffixText: 'nilai MQ-2',
              ),
              validator: (value) {
                final parsed = int.tryParse(value ?? '');
                if (parsed == null || parsed < 1 || parsed > 4095) {
                  return 'Masukkan angka 1 sampai 4095';
                }

                final warning = int.tryParse(_warningController.text);
                if (warning != null && parsed <= warning) {
                  return 'Harus lebih besar dari Warning';
                }
                return null;
              },
            ),
            const SizedBox(height: 8),
            const Text(
              'Warning harus lebih kecil daripada Danger.',
              style: TextStyle(fontSize: 11, color: Colors.black54),
            ),
          ],
        ),
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.of(context).pop(),
          child: const Text('Batal'),
        ),
        FilledButton(
          style: FilledButton.styleFrom(backgroundColor: _accent),
          onPressed: _save,
          child: const Text('Simpan'),
        ),
      ],
    );
  }
}

  const HistoryScreen({super.key, required this.repository});
  final ReadingRepository repository;
  @override
  State<HistoryScreen> createState() => _HistoryScreenState();
}

class _HistoryScreenState extends State<HistoryScreen> {
  int _selectedCount = 20;
  String _selectedHistoryDevice = 'Semua perangkat';
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: _background,
      appBar: AppBar(
        backgroundColor: _background,
        elevation: 0,
        centerTitle: true,
        leading: IconButton(
          icon: const Icon(Icons.arrow_back, color: Colors.black87),
          onPressed: () => Navigator.of(context).pop(),
        ),
        title: const Text(
          'Riwayat Suhu',
          style: TextStyle(
            fontSize: 15,
            fontWeight: FontWeight.bold,
            color: Colors.black87,
          ),
        ),
      ),
      body: StreamBuilder<List<Reading>>(
        stream: widget.repository.getReadings(),
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting &&
              !snapshot.hasData) {
            return const Center(
              child: CircularProgressIndicator(color: _accent),
            );
          }
          if (snapshot.hasError) {
            return _message(
              'Riwayat tidak dapat dibaca',
              'Periksa koneksi internet dan pengaturan Firebase.\n${snapshot.error}',
            );
          }
          final allReadings = snapshot.data ?? [];
          if (allReadings.isEmpty) {
            return _message(
              'Belum ada riwayat',
              'Data sensor akan tampil setelah tersedia di Firebase.',
            );
          }
          final deviceIds = allReadings
              .map((reading) => reading.deviceId.trim().isEmpty ? 'Perangkat tanpa ID' : reading.deviceId.trim())
              .toSet()
              .toList()
            ..sort();
          final selectedDevice = _selectedHistoryDevice == 'Semua perangkat' || deviceIds.contains(_selectedHistoryDevice)
              ? _selectedHistoryDevice

class _TemperatureChartPainter extends CustomPainter {
  _TemperatureChartPainter(this.values);
  final List<double> values;
  @override
  void paint(Canvas canvas, Size size) {
    if (values.length < 2 || size.isEmpty) return;
    final minValue = values.reduce((a, b) => a < b ? a : b);
    final maxValue = values.reduce((a, b) => a > b ? a : b);
    final range = (maxValue - minValue).abs() < 0.1 ? 1.0 : maxValue - minValue;
    const padding = 12.0;
    final chartHeight = size.height - padding * 2;
    final gridPaint = Paint()
      ..color = const Color(0xFFD8C9C6)
      ..strokeWidth = 1;
    for (var i = 0; i < 4; i++) {
      final y = padding + chartHeight * i / 3;
      canvas.drawLine(Offset(0, y), Offset(size.width, y), gridPaint);
    }
    final path = Path();
    for (var i = 0; i < values.length; i++) {
      final x = size.width * i / (values.length - 1);
      final y = padding + chartHeight * (1 - (values[i] - minValue) / range);
      if (i == 0) {
        path.moveTo(x, y);
      } else {
        path.lineTo(x, y);
      }
    }
    final linePaint = Paint()
      ..color = _accent
      ..strokeWidth = 2.0
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;
    canvas.drawPath(path, linePaint);
  }

  @override
  bool shouldRepaint(covariant _TemperatureChartPainter oldDelegate) {
    if (oldDelegate.values.length != values.length) return true;
    for (var i = 0; i < values.length; i++) {
      if (oldDelegate.values[i] != values[i]) return true;
    }
    return false;
  }
}

Color _statusColor(String status) {
  if (status == 'DANGER' || status == 'DANGER') return Colors.red;
  if (status == 'WARNING' || status == 'WARNING') return Colors.orange;
  return const Color(0xFF4CAF50);
}

String _statusLabel(String status) {
  if (status == 'DANGER' || status == 'DANGER') return 'Danger';
  if (status == 'WARNING' || status == 'WARNING') return 'Warning';
  return 'Normal';
}

String _formatDateTime(DateTime dateTime) {
  return '${dateTime.day.toString().padLeft(2, '0')}/'
      '${dateTime.month.toString().padLeft(2, '0')}/'
      '${dateTime.year} '
      '${dateTime.hour.toString().padLeft(2, '0')}:'
      '${dateTime.minute.toString().padLeft(2, '0')}:'
      '${dateTime.second.toString().padLeft(2, '0')}';
}

String _formatTime(DateTime dateTime) {
  return '${dateTime.hour.toString().padLeft(2, '0')}.'
      '${dateTime.minute.toString().padLeft(2, '0')}';
}
